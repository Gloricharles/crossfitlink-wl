/** * CrossFit Link - Suivi Weightlifting * Google Apps Script - déployer comme Web App * Exécution : Moi | Accès : Tout le monde */ const SHEET_NAME = 'Logs'; function doGet(e) { const p = e.parameter || {}; let result; try { switch (p.action) { case 'write': result = writeEntry(p); break; case 'read': result = readAthlete(p.name); break; case 'roster': result = getRoster(); break; case 'all': result = getAllData(); break; default: result = { status: 'error', message: 'Action inconnue' }; } } catch (err) { result = { status: 'error', message: err.toString() }; } const json = JSON.stringify(result); // JSONP : si un callback est fourni, on renvoie du JavaScript exécutable // (contourne les restrictions CORS des appels cross-domain) if (p.callback) { return ContentService .createTextOutput(p.callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let s = ss.getSheetByName(SHEET_NAME); if (!s) { s = ss.insertSheet(SHEET_NAME); s.appendRow(['Timestamp', 'Nom', 'Semaine', 'Main lift (kg)', 'Accessoire (kg)']); s.setFrozenRows(1); } return s; } function writeEntry(p) { getSheet().appendRow([ new Date().toISOString(), (p.name || '').trim(), p.week || '', p.value !== undefined && p.value !== '' ? Number(p.value) : '', p.secondary !== undefined && p.secondary !== '' ? Number(p.secondary) : '' ]); return { status: 'ok' }; } function readAthlete(name) { if (!name) return { status: 'error', message: 'Nom manquant' }; const rows = getSheet().getDataRange().getValues(); const key = name.trim().toLowerCase(); const weeks = {}; for (let i = 1; i < rows.length; i++) { const r = rows[i]; if ((r[1] || '').toString().trim().toLowerCase() !== key) continue; const wid = (r[2] || '').toString(); if (!wid) continue; weeks[wid] = { value: r[3] !== '' ? r[3] : null, secondary: r[4] !== '' ? r[4] : null, ts: r[0] ? r[0].toString() : '' }; } return { status: 'ok', name: name.trim(), weeks: weeks }; } function getRoster() { const rows = getSheet().getDataRange().getValues(); const seen = {}; const names = []; for (let i = 1; i < rows.length; i++) { const name = (rows[i][1] || '').toString().trim(); const k = name.toLowerCase(); if (name && !seen[k]) { seen[k] = true; names.push(name); } } names.sort(function(a, b){ return a.localeCompare(b, 'fr'); }); return { status: 'ok', names: names }; } function getAllData() { const rows = getSheet().getDataRange().getValues(); const map = {}; for (let i = 1; i < rows.length; i++) { const r = rows[i]; const name = (r[1] || '').toString().trim(); if (!name) continue; const k = name.toLowerCase(); const wid = (r[2] || '').toString(); if (!wid) continue; if (!map[k]) map[k] = { name: name, weeks: {} }; map[k].weeks[wid] = { value: r[3] !== '' ? r[3] : null, secondary: r[4] !== '' ? r[4] : null }; } const athletes = Object.keys(map).map(function(k){ return map[k]; }); athletes.sort(function(a, b){ return a.name.localeCompare(b.name, 'fr'); }); return { status: 'ok', athletes: athletes }; }
